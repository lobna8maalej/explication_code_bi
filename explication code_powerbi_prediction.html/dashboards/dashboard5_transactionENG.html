DATABASE

Transaction Storage with MySQL

After data preparation, part of the project uses MySQL to store prediction results.

The connection to MySQL is established using the mysql2/promise module in Node.js.

transactionDb.js 
const mysql = require("mysql2/promise"); 
 
const dbConfig = { 
 
    host: "localhost", 
 
    user: "root", 
 
    password: "", 
 
    database: "POWERBI_PREDICTION" 
 
}; 
 
async function getConnection() { 
 
    return await mysql.createConnection(dbConfig); 
 
} 
 
module.exports = { getConnection };

mysql2/promise

This library allows Node.js to communicate with MySQL using asynchronous functions.

dbConfig

This object contains the information required to establish a connection to the database.

getConnection()

This function opens a connection to the database.

module.exports

Allows the function to be exported so that it can be used in other Node.js files.

BACKEND

Communication Between Node.js and Python

Node.js acts as an intermediary between the application and the Python program.

The transaction.js service uses the child_process module to launch the Python program.

transaction.js 
const { spawn } = require("child_process"); 
const path = require("path"); 
 
function runPython(mode, value) { 
 
    return new Promise((resolve, reject) => { 
 
        const scriptPath = path.join( 
            __dirname, 
            "../../python/app.py" 
        ); 
 
        const py = spawn("python", [ 
            scriptPath, 
            mode, 
            value.toString() 
        ]); 
 
        let output = ""; 
 
        py.stdout.on( 
            "data", 
            data => output += data.toString() 
        ); 
 
        py.stderr.on( 
            "data", 
            err => console.error( 
                "PY ERROR:", 
                err.toString() 
            ) 
        ); 
 
        py.on("close", (code) => { 
 
            if (code !== 0) 
                return reject("Python error"); 
 
            resolve(output.trim()); 
 
        }); 
 
    }); 
 
} 
 
module.exports = runPython;

REST API

Express Prediction Route

POST /transactions

The Express route receives the information required for the prediction.

It then calls the Python service, retrieves the prediction and saves the result in MySQL.

transactionRoute.js 
router.post("/", async (req, res) => { 
 
    let connection; 
 
    try { 
 
        const { mode, value } = req.body; 
 
        if (!mode || value === undefined) { 
 
            return res.status(400).json({ 
                error: "mode and value are required" 
            }); 
 
        } 
 
        connection = await getConnection(); 
 
        const result = 
            await runPython(mode, value); 
 
        console.log( 
            "PYTHON RESULT:", 
            result 
        ); 
 
        const data = 
            typeof result === "string" 
                ? JSON.parse(result) 
                : result; 
 
        const prediction = 
            data?.prediction ?? null; 
 
        if (prediction === null) { 
 
            return res.status(400).json({ 
                error: "Invalid prediction from Python", 
                debug: data 
            }); 
 
        } 
 
        await connection.execute( 
            `INSERT INTO transactions 
             (mode, value, prediction) 
             VALUES (?, ?, ?)`, 
            [ 
                mode, 
                value, 
                prediction 
            ] 
        ); 
 
        return res.json({ 
 
            message: "Prediction saved", 
 
            mode, 
 
            value, 
 
            prediction 
 
        }); 
 
    } catch (err) { 
 
        console.error(err); 
 
        return res.status(500).json({ 
            error: err.toString() 
        }); 
 
    } finally { 
 
        if (connection) 
            await connection.end(); 
 
    } 
 
});
