const mysql = require("mysql2/promise");

const dbConfig = {

    host: "localhost",

    user: "root",

    password: "",

    database: "POWERBI_PREDICTION"

};

async function getConnection() {

    return await mysql.createConnection(dbConfig);

}

module.exports = { getConnection };

mysql2/promise

Esta biblioteca permite que o Node.js se comunique com o MySQL utilizando funções assíncronas.

Este objeto contém as informações necessárias para estabelecer uma conexão com o banco de dados.

MODELO NCG

Previsão do número da conta geral

Quando o parâmetro mode é ncg, o programa carrega o modelo NCG_model.pkl.

Python 
if mode == "ncg":

    model_path = os.path.join(
        BASE_DIR,
        "NCG_model.pkl"
    )

    with open(model_path, "rb") as f:
        model = pickle.load(f)

        "input": value,
        "prediction": float(prediction[0])

3. Carregamento

O Pickle recupera o modelo de Machine Learning previamente treinado.

4. Previsão

O método model.predict() produz o valor previsto.

BACKEND

Comunicação entre Node.js e Python

O Node.js atua como intermediário entre a aplicação e o programa Python.

O serviço transaction.js utiliza o módulo child_process para executar o programa Python.

transaction.js 
const { spawn } = require("child_process");
const path = require("path");

function runPython(mode, value) {

    return new Promise((resolve, reject) => {

        const scriptPath = path.join(
            __dirname,
            "../../python/app.py"
        );

        const py = spawn("python", [
            scriptPath,
            mode,
            value.toString()
        ]);

        let output = "";

        py.stdout.on(
            "data",
            data => output += data.toString()
        );

        py.stderr.on(
            "data",
            err => console.error(
                "PY ERROR:",
                err.toString()
            )
        );

        py.on("close", (code) => {

            if (code !== 0)
                return reject("Python error");

            resolve(output.trim());

        });

    });

}

module.exports = runPython;

REST API

Rota de previsão do Express

POST /transactions

A rota Express recebe as informações necessárias para realizar a previsão.

Em seguida, chama o serviço Python, recupera a previsão e salva o resultado no MySQL.

transactionRoute.js 
router.post("/", async (req, res) => {

    let connection;

    try {

        const { mode, value } = req.body;

        if (!mode || value === undefined) {

            return res.status(400).json({
                error: "mode and value are required"
            });

        }

        connection = await getConnection();

        const result =
            await runPython(mode, value);

        console.log(
            "PYTHON RESULT:",
            result
        );

        const data =
            typeof result === "string"
                ? JSON.parse(result)
                : result;

        const prediction =
            data?.prediction ?? null;

        if (prediction === null) {

            return res.status(400).json({
                error: "Invalid prediction from Python",
                debug: data
            });

        }

        await connection.execute(
            `INSERT INTO transactions
             (mode, value, prediction)
             VALUES (?, ?, ?)`,
            [
                mode,
                value,
                prediction
            ]
        );

        return res.json({

            message: "Prediction saved",

            mode,

            value,

            prediction

        });

    } catch (err) {

        console.error(err);

        return res.status(500).json({
            error: err.toString()
        });

    } finally {

        if (connection)
            await connection.end();

    }

});
