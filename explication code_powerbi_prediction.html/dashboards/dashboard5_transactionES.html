BASE DE DATOS

Almacenamiento de transacciones con MySQL

Después de la preparación de los datos, una parte del proyecto utiliza MySQL para almacenar los resultados de las predicciones.

La conexión con MySQL se establece utilizando el módulo mysql2/promise en Node.js.

transactionDb.js 
const mysql = require("mysql2/promise"); 
 
const dbConfig = { 
 
    host: "localhost", 
 
    user: "root", 
 
    password: "", 
 
    database: "POWERBI_PREDICTION" 
 
}; 
 
async function getConnection() { 
 
    return await mysql.createConnection(dbConfig); 
 
} 
 
module.exports = { getConnection };

mysql2/promise

Esta biblioteca permite a Node.js comunicarse con MySQL utilizando funciones asíncronas.

dbConfig

Este objeto contiene la información necesaria para establecer una conexión con la base de datos.

getConnection()

Esta función abre una conexión con la base de datos.

module.exports

Permite exportar la función para que pueda utilizarse en otros archivos de Node.js.

MODELO NCG

Predicción del número de cuenta general

Cuando el parámetro mode es ncg, el programa carga el modelo NCG_model.pkl.

Python 
if mode == "ncg": 
 
    model_path = os.path.join( 
        BASE_DIR, 
        "NCG_model.pkl" 
    ) 
 
    with open(model_path, "rb") as f: 
        model = pickle.load(f) 
 
    prediction = model.predict([[value]]) 
 
    result = { 
        "model": "ncg", 
        "input": value, 
        "prediction": float(prediction[0]) 
    }

1. Verificación del modo

El programa comprueba si el modo recibido corresponde a ncg.

2. Ruta del modelo

La ruta hacia NCG_model.pkl se construye utilizando os.path.join().

3. Carga

Pickle recupera el modelo de Machine Learning previamente entrenado.

4. Predicción

El método model.predict() produce el valor predicho.

BACKEND

Comunicación entre Node.js y Python

Node.js actúa como intermediario entre la aplicación y el programa Python.

El servicio transaction.js utiliza el módulo child_process para ejecutar el programa Python.

transaction.js 
const { spawn } = require("child_process"); 
const path = require("path"); 
 
function runPython(mode, value) { 
 
    return new Promise((resolve, reject) => { 
 
        const scriptPath = path.join( 
            __dirname, 
            "../../python/app.py" 
        ); 
 
        const py = spawn("python", [ 
            scriptPath, 
            mode, 
            value.toString() 
        ]); 
 
        let output = ""; 
 
        py.stdout.on( 
            "data", 
            data => output += data.toString() 
        ); 
 
        py.stderr.on( 
            "data", 
            err => console.error( 
                "PY ERROR:", 
                err.toString() 
            ) 
        ); 
 
        py.on("close", (code) => { 
 
            if (code !== 0) 
                return reject("Python error"); 
 
            resolve(output.trim()); 
 
        }); 
 
    }); 
 
} 
 
module.exports = runPython;

REST API

Ruta de predicción de Express

POST /transactions

La ruta Express recibe la información necesaria para realizar la predicción.

A continuación, llama al servicio Python, recupera la predicción y guarda el resultado en MySQL.

transactionRoute.js 
router.post("/", async (req, res) => { 
 
    let connection; 
 
    try { 
 
        const { mode, value } = req.body; 
 
        if (!mode || value === undefined) { 
 
            return res.status(400).json({ 
                error: "mode and value are required" 
            }); 
 
        } 
 
        connection = await getConnection(); 
 
        const result = 
            await runPython(mode, value); 
 
        console.log( 
            "PYTHON RESULT:", 
            result 
        ); 
 
        const data = 
            typeof result === "string" 
                ? JSON.parse(result) 
                : result; 
 
        const prediction = 
            data?.prediction ?? null; 
 
        if (prediction === null) { 
 
            return res.status(400).json({ 
                error: "Invalid prediction from Python", 
                debug: data 
            }); 
 
        } 
 
        await connection.execute( 
            `INSERT INTO transactions 
             (mode, value, prediction) 
             VALUES (?, ?, ?)`, 
            [ 
                mode, 
                value, 
                prediction 
            ] 
        ); 
 
        return res.json({ 
 
            message: "Prediction saved", 
 
            mode, 
 
            value, 
 
            prediction 
 
        }); 
 
    } catch (err) { 
 
        console.error(err); 
 
        return res.status(500).json({ 
            error: err.toString() 
        }); 
 
    } finally { 
 
        if (connection) 
            await connection.end(); 
 
    } 
 
});
